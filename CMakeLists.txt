cmake_minimum_required(VERSION 3.24)
project(auralyn_obs VERSION 0.1.0 LANGUAGES CXX)
option(AURALYN_SINC48_POC "Experimental 48k/88.2k/96k/176.4k/192k-to-48k sinc interpolation; other rates retain linear interpolation" OFF)
if(AURALYN_SINC48_POC)
  add_compile_definitions(AURALYN_SINC48_POC=1)
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This prototype requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OBS_SOURCE_DIR "${CMAKE_SOURCE_DIR}/build/obs-studio" CACHE PATH "Pinned OBS 32.2.2 source checkout")
set(OBS_BIN_DIR "C:/Program Files/obs-studio/bin/64bit" CACHE PATH "OBS 32.2.2 runtime; no files are modified")
if(EXISTS "${OBS_SOURCE_DIR}/.git")
  execute_process(COMMAND git -C "${OBS_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE obs_commit OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT obs_commit STREQUAL "ba2f32bdf791005443988a4955e963663e16b1ed")
    message(FATAL_ERROR "The OBS SDK commit does not match protocol.lock.json")
  endif()
endif()
file(READ "${OBS_SOURCE_DIR}/libobs/obs-config.h" obs_config)
if(NOT obs_config MATCHES "LIBOBS_API_MAJOR_VER 32" OR NOT obs_config MATCHES "LIBOBS_API_MINOR_VER 2" OR NOT obs_config MATCHES "LIBOBS_API_PATCH_VER 2")
  message(FATAL_ERROR "Use the pinned OBS 32.2.2 headers")
endif()
set(OBS_RELEASE_CANDIDATE 0)
set(OBS_BETA 0)
configure_file("${OBS_SOURCE_DIR}/libobs/obsconfig.h.in" "${CMAKE_BINARY_DIR}/include/obsconfig.h")
get_filename_component(msvc_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(DUMPBIN dumpbin HINTS "${msvc_bin}" REQUIRED)
find_program(MSVC_LIB lib HINTS "${msvc_bin}" REQUIRED)
execute_process(COMMAND "${DUMPBIN}" /exports "${OBS_BIN_DIR}/obs.dll" OUTPUT_VARIABLE exports RESULT_VARIABLE result)
if(NOT result EQUAL 0)
  message(FATAL_ERROR "Cannot inspect the installed OBS DLL")
endif()
string(REGEX MATCHALL "[\r\n] +[0-9]+ +[A-Fa-f0-9]+ +[A-Fa-f0-9]+ +[A-Za-z_?][^ \r\n]*" symbols "${exports}")
set(def "LIBRARY obs.dll\nEXPORTS\n")
foreach(symbol IN LISTS symbols)
  string(REGEX REPLACE ".* +" "" name "${symbol}")
  string(APPEND def "${name}\n")
endforeach()
file(WRITE "${CMAKE_BINARY_DIR}/obs.def" "${def}")
execute_process(COMMAND "${MSVC_LIB}" /nologo /machine:x64 "/def:${CMAKE_BINARY_DIR}/obs.def" "/out:${CMAKE_BINARY_DIR}/obs.lib" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
  message(FATAL_ERROR "Cannot generate local OBS import library")
endif()
add_library(obs_sdk INTERFACE)
target_include_directories(obs_sdk SYSTEM INTERFACE "${OBS_SOURCE_DIR}/libobs" "${CMAKE_BINARY_DIR}/include")
target_link_libraries(obs_sdk INTERFACE "${CMAKE_BINARY_DIR}/obs.lib")
add_library(auralyn-obs MODULE src/plugin.cpp)
target_link_libraries(auralyn-obs PRIVATE obs_sdk)
target_compile_options(auralyn-obs PRIVATE /W4 /utf-8 /permissive-)
enable_testing()
add_executable(pcm_tests tests/pcm_tests.cpp)
target_include_directories(pcm_tests PRIVATE src)
target_compile_options(pcm_tests PRIVATE /W4 /utf-8)
target_compile_definitions(pcm_tests PRIVATE FIXTURE_PATH="${CMAKE_SOURCE_DIR}/protocol/obs-pcm-v1.json")
add_test(NAME pcm_contract COMMAND pcm_tests)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME protocol_snapshot COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/verify_protocol.py")
add_executable(obs_smoke tests/obs_smoke.cpp)
target_link_libraries(obs_smoke PRIVATE obs_sdk)
target_compile_options(obs_smoke PRIVATE /W4 /utf-8)
